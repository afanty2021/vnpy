# P1-1: 错误处理统一方案

## 问题描述

当前代码中错误处理方式不一致：

| 模块 | 错误处理方式 | 问题 |
|------|-------------|------|
| `database.py` | `logger.error()` | 正确 |
| `service.py` | `print()` | 无法控制日志级别 |
| `rpc_qmt_adapter.py` | 混合使用 `logger` 和 `print()` | 不一致 |

**问题**：
1. 无法统一控制日志输出级别
2. 无法将错误日志写入文件
3. 生产环境调试困难
4. 日志格式不统一

## 修复方案

### 方案设计

创建统一的日志配置模块，所有模块使用相同的logger配置。

### 实现步骤

#### 步骤1: 创建统一日志模块

创建 `vnpy_china_config/logging_config.py`：

```python
"""
统一日志配置模块

为所有vnpy_china_*模块提供统一的日志配置。
"""

import logging
import sys
from pathlib import Path
from typing import Optional
from logging.handlers import RotatingFileHandler, TimedRotatingFileHandler


# 日志格式
LOG_FORMAT = "%(asctime)s - %(name)s - %(levelname)s - %(message)s"
LOG_FORMAT_DETAILED = "%(asctime)s - %(name)s - %(levelname)s - [%(filename)s:%(lineno)d] - %(message)s"
DATE_FORMAT = "%Y-%m-%d %H:%M:%S"

# 模块日志名称
MODULE_NAMES = [
    "vnpy_china_config",
    "vnpy_china_data",
    "vnpy_china_rules",
    "vnpy_china_analysis",
    "vnpy_china_backtest",
    "vnpy_china_capital",
    "vnpy_china_ml",
]


def setup_logging(
    level: str = "INFO",
    log_file: Optional[Path] = None,
    console_enabled: bool = True,
    max_bytes: int = 10 * 1024 * 1024,  # 10MB
    backup_count: int = 5,
    detailed_format: bool = False,
) -> None:
    """配置统一的日志系统

    Args:
        level: 日志级别 (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        log_file: 日志文件路径（可选）
        console_enabled: 是否输出到控制台
        max_bytes: 单个日志文件最大字节数
        backup_count: 保留的日志文件数量
        detailed_format: 是否使用详细格式（包含文件名和行号）

    Example:
        ```python
        from vnpy_china_config.logging_config import setup_logging

        # 基本配置
        setup_logging(level="INFO")

        # 完整配置
        setup_logging(
            level="DEBUG",
            log_file=Path("logs/vnpy_china.log"),
            console_enabled=True,
            detailed_format=True
        )
        ```
    """
    # 转换日志级别
    log_level = getattr(logging, level.upper(), logging.INFO)

    # 选择日志格式
    log_format = LOG_FORMAT_DETAILED if detailed_format else LOG_FORMAT
    formatter = logging.Formatter(log_format, datefmt=DATE_FORMAT)

    # 获取根日志器
    root_logger = logging.getLogger()
    root_logger.setLevel(log_level)

    # 清除现有处理器
    root_logger.handlers.clear()

    # 添加控制台处理器
    if console_enabled:
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.setLevel(log_level)
        console_handler.setFormatter(formatter)
        root_logger.addHandler(console_handler)

    # 添加文件处理器
    if log_file:
        # 确保日志目录存在
        log_file.parent.mkdir(parents=True, exist_ok=True)

        # 使用RotatingFileHandler按大小切割日志
        file_handler = RotatingFileHandler(
            log_file,
            maxBytes=max_bytes,
            backupCount=backup_count,
            encoding="utf-8",
        )
        file_handler.setLevel(log_level)
        file_handler.setFormatter(formatter)
        root_logger.addHandler(file_handler)

    # 配置第三方库的日志级别（减少噪音）
    logging.getLogger("pymysql").setLevel(logging.WARNING)
    logging.getLogger("urllib3").setLevel(logging.WARNING)
    logging.getLogger("requests").setLevel(logging.WARNING)


def get_logger(name: str) -> logging.Logger:
    """获取模块专用logger

    Args:
        name: 模块名称（通常使用 __name__）

    Returns:
        logging.Logger: 配置好的logger实例

    Example:
        ```python
        from vnpy_china_config.logging_config import get_logger

        logger = get_logger(__name__)
        logger.info("这是一条信息日志")
        logger.error("这是一条错误日志", exc_info=True)
        ```
    """
    return logging.getLogger(name)


def get_logger_for_module(module_name: str) -> logging.Logger:
    """为vnpy_china模块获取logger

    Args:
        module_name: 模块名称（如 "vnpy_china_data"）

    Returns:
        logging.Logger: 配置好的logger实例
    """
    return logging.getLogger(module_name)


class LoggerMixin:
    """Logger混入类

    为类提供logger属性。

    Example:
        ```python
        from vnpy_china_config.logging_config import LoggerMixin

        class MyService(LoggerMixin):
            def __init__(self):
                super().__init__()
                self.logger.info("服务初始化")
        ```
    """

    @property
    def logger(self) -> logging.Logger:
        """获取当前类的logger"""
        return logging.getLogger(self.__class__.__module__)


def log_function_call(logger: Optional[logging.Logger] = None):
    """装饰器：记录函数调用

    Args:
        logger: logger实例，如果为None则使用函数所在模块的logger

    Example:
        ```python
        from vnpy_china_config.logging_config import log_function_call

        @log_function_call()
        def my_function(x, y):
            return x + y
        ```
    """
    import functools

    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            func_logger = logger or logging.getLogger(func.__module__)
            func_logger.debug(f"调用 {func.__name__}: args={args}, kwargs={kwargs}")
            try:
                result = func(*args, **kwargs)
                func_logger.debug(f"{func.__name__} 返回: {result}")
                return result
            except Exception as e:
                func_logger.error(f"{func.__name__} 抛出异常: {e}", exc_info=True)
                raise
        return wrapper
    return decorator


class ExceptionHandler:
    """统一异常处理器

    提供一致的异常处理和日志记录。

    Example:
        ```python
        from vnpy_china_config.logging_config import ExceptionHandler

        handler = ExceptionHandler(logger)

        try:
            risky_operation()
        except Exception as e:
            handler.handle(e, context="执行风险操作", reraise=False)
        ```
    """

    def __init__(self, logger: logging.Logger):
        """初始化异常处理器

        Args:
            logger: logger实例
        """
        self.logger = logger

    def handle(
        self,
        exception: Exception,
        context: str = "",
        level: str = "ERROR",
        reraise: bool = True,
    ) -> None:
        """处理异常

        Args:
            exception: 异常对象
            context: 上下文信息
            level: 日志级别 (ERROR, CRITICAL)
            reraise: 是否重新抛出异常
        """
        log_func = getattr(self.logger, level.lower(), self.logger.error)

        context_msg = f"[{context}] " if context else ""
        log_msg = f"{context_msg}{type(exception).__name__}: {exception}"

        log_func(log_msg, exc_info=True)

        if reraise:
            raise

    def handle_and_return(
        self,
        exception: Exception,
        context: str = "",
        default_return=None,
    ):
        """处理异常并返回默认值

        Args:
            exception: 异常对象
            context: 上下文信息
            default_return: 默认返回值

        Returns:
            default_return
        """
        self.handle(exception, context=context, reraise=False)
        return default_return
```

    """MySQL数据库层（线程安全连接池版本）"""

    def __init__(self, host: str, port: int, user: str, password: str, database: str, charset: str = "utf8mb4"):
        """初始化数据库连接池"""
        self.config = {
            "host": host,
            "port": port,
            "user": user,
            "password": password,
            "database": database,
            "charset": charset,
        }
        self._pool: Optional[PooledDB] = None
        self._lock = Lock()
        self._connected = False

        """初始化数据库连接池"""
        if not PYMYSQL_AVAILABLE:
            logger.error("pymysql 或 DBUtils 未安装")
            return False

            with self._lock:
                if self._pool is not None:
                    return True

                    creator=pymysql,
                    maxconnections=self.pool_size + self.max_overflow,
                    mincached=self.pool_size,
                    maxcached=self.pool_size,
                    maxshared=0,
                    blocking=True,
                    maxusage=0,
                    setsession=[],
                    ping=1,
                    reset=True,
                    **self.config
                )

                logger.info(
                    f"MySQL连接池创建成功: {self.config['host']}:{self.config['port']}/"
                    f"{self.config['database']} (池大小: {self.pool_size})"
                )
                return True

        except (OperationalError, DatabaseError) as e:
            self._connected = False
            logger.warning(f"MySQL连接池创建失败: {e}", exc_info=True)
            return False
        except Exception as e:
            self._connected = False
            logger.warning(f"MySQL连接池异常: {e}", exc_info=True)
            return False

        self,

"""
vnpy_china_config - 配置管理模块

提供统一的配置管理和日志配置功能。
"""

from .loader import ConfigManager
from .base import BaseConfig, Environment
from .global_config import GlobalConfig
from .logging_config import setup_logging, get_logger

# 初始化日志（使用默认配置）
setup_logging(level="INFO")


def initialize_system(config_path: str = None):
    """初始化系统配置和日志

    Args:
        config_path: 配置文件路径（可选）

    Returns:
        GlobalConfig: 全局配置对象

    Example:
        ```python
        from vnpy_china_config import initialize_system

        config = initialize_system()
        # 现在可以正常使用logger了
        ```
    """
    manager = ConfigManager()
    if config_path:
        manager.set_config_path(config_path)

    config = manager.load_global_config()

    # 根据配置设置日志
    config.setup_logging()

    return config


__all__ = [
    "ConfigManager",
    "BaseConfig",
    "Environment",
    "GlobalConfig",
    "setup_logging",
    "get_logger",
    "initialize_system",
]
